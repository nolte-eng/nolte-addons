function parseNwtfvHtml(fi,html){
 const doc=new DOMParser().parseFromString(html,'text/html');
 const rows=[...doc.querySelectorAll('table tr')];
 const parsed=[];
 for(const row of rows){
  const cells=[...row.querySelectorAll('td')];
  if(cells.length<4)continue;
  const num=(cells[0].textContent||'').trim();
  if(!/^[1-8]$/.test(num))continue;
  let resultIndex=cells.findIndex(c=>/^\s*[0-2]\s*:\s*[0-2]\s*$/.test(c.textContent||''));
  if(resultIndex<0)continue;
  const clean=c=>[...c.querySelectorAll('a')].map(a=>a.textContent.trim()).filter(Boolean).join(' / ') || c.textContent.trim();
  const home=clean(cells[Math.max(1,resultIndex-1)]);
  const away=clean(cells[Math.min(cells.length-1,resultIndex+1)]);
  const m=(cells[resultIndex].textContent||'').match(/([0-2])\s*:\s*([0-2])/);
  parsed.push({idx:Number(num)-1,home,away,winner:m?(Number(m[1])>Number(m[2])?'home':Number(m[2])>Number(m[1])?'away':''):''});
 }
 return applyParsed(fi,parsed);
}
function parseNwtfvText(fi,text){
 const lines=text.split(/\r?\n/).map(x=>x.trim()).filter(Boolean);
 const parsed=[];
 for(let n=1;n<=8;n++){
  const ix=lines.findIndex((x,i)=>x===String(n) && i<lines.length-2);
  if(ix<0)continue;
  let scoreIx=-1;
  for(let j=ix+1;j<Math.min(lines.length,ix+8);j++){if(/^[0-2]\s*:\s*[0-2]$/.test(lines[j])){scoreIx=j;break}}
  if(scoreIx<0)continue;
  const before=lines.slice(ix+1,scoreIx).filter(x=>!/^Image$/i.test(x));
  const after=lines.slice(scoreIx+1,scoreIx+5).filter(x=>!/^Image$/i.test(x)&&!/^[1-8]$/.test(x));
  const m=lines[scoreIx].match(/([0-2])\s*:\s*([0-2])/);
  parsed.push({idx:n-1,home:before.join(' / '),away:after.join(' / '),winner:Number(m[1])>Number(m[2])?'home':Number(m[2])>Number(m[1])?'away':''});
 }
 return applyParsed(fi,parsed);
}
function applyParsed(fi,parsed){
 if(!parsed.length)return 0;
 const f=R().fixtures[fi];
 const existing=f.games.some(g=>g.homePlayers||g.awayPlayers||g.winner);
 if(existing&&!confirm('Es sind bereits manuelle Aufstellungs-/Ergebnisdaten vorhanden. Mit den NWTFV-Daten überschreiben?'))return -1;
 parsed.forEach(p=>{if(f.games[p.idx]){f.games[p.idx].homePlayers=p.home;f.games[p.idx].awayPlayers=p.away;f.games[p.idx].winner=p.winner||''}});
 save();return parsed.length;
}

}
