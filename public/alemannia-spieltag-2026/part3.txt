function start(ti,fi){
 const s=document.getElementById('gsel'+ti);if(!s||s.value===''){alert('Bitte ein Spiel auswählen.');return}
 const gi=Number(s.value);if(gameDone(R().fixtures[fi].games[gi])){alert('Spiel bereits beendet.');return}
 R().tables[ti]={fixture:fi,game:gi};save();
}
function finish(ti){
 const t=R().tables[ti];if(t.fixture===null)return;
 const g=R().fixtures[t.fixture].games[t.game];
 if(!gameDone(g)){alert('Bitte zuerst in der Aufstellung auswählen, wer das Spiel gewonnen hat.');return}
 R().tables[ti]={fixture:null,game:null};save();
}
